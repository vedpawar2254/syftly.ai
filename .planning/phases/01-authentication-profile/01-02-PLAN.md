```javascript
import Profile from '../models/Profile.js';

async function createProfile(userId, profileData) {
  const existingProfile = await Profile.findOne({ userId });
  if (existingProfile) {
    throw new Error('Profile already exists');
  }

  const profile = new Profile({
    userId,
    ...profileData
  });

  await profile.save();
  return profile;
}

async function updateProfile(userId, profileData) {
  const profile = await Profile.findOne({ userId });

  if (!profile) {
    return createProfile(userId, profileData);
  }

  Object.assign(profile, profileData);
  await profile.save();

  return profile;
}

async function getProfile(userId) {
  const profile = await Profile.findOne({ userId }).populate('userId', 'email isEmailVerified');

  if (!profile) {
    return null;
  }

  return {
    id: profile._id,
    displayName: profile.displayName,
    bio: profile.bio,
    avatarUrl: profile.avatarUrl,
    synthesisPreferences: profile.synthesisPreferences,
    email: profile.userId.email,
    isEmailVerified: profile.userId.isEmailVerified
  };
}

async function updatePreferences(userId, preferences) {
  const profile = await Profile.findOne({ userId });

  if (!profile) {
    throw new Error('Profile not found. Please create a profile first.');
  }

  profile.synthesisPreferences = {
    ...profile.synthesisPreferences,
    ...preferences
  };

  await profile.save();
  return profile.synthesisPreferences;
}

export default {
  createProfile,
  updateProfile,
  getProfile,
  updatePreferences
};
```

```javascript
import multer from 'multer';
import path from 'path';
import fs from 'fs';

const uploadDir = 'uploads/avatars';

if (!fs.existsSync(uploadDir)) {
  fs.mkdirSync(uploadDir, { recursive: true });
}

const storage = multer.diskStorage({
  destination: (req, file, cb) => {
    cb(null, uploadDir);
  },
  filename: (req, file, cb) => {
    const uniqueSuffix = Date.now() + '-' + Math.round(Math.random() * 1E9);
    cb(null, 'avatar-' + req.user.userId + '-' + uniqueSuffix + path.extname(file.originalname));
  }
});

const fileFilter = (req, file, cb) => {
  const allowedTypes = ['image/jpeg', 'image/png', 'image/webp'];
  if (allowedTypes.includes(file.mimetype)) {
    cb(null, true);
  } else {
    cb(new Error('Invalid file type. Only JPEG, PNG, and WebP allowed.'), false);
  }
};

const upload = multer({
  storage,
  fileFilter,
  limits: {
    fileSize: 5 * 1024 * 1024
  }
});

export default upload;
```

```javascript
const router = express.Router();

router.post('/avatar', requireAuth, upload.single('avatar'), async (req, res) => {
  try {
    if (!req.file) {
      return error(res, 'No file uploaded', 400);
    }

    const userId = req.user.userId;
    const avatarUrl = `/uploads/avatars/${req.file.filename}`;

    let profile = await Profile.findOne({ userId });

    if (!profile) {
      profile = new Profile({ userId, avatarUrl });
    } else {
      profile.avatarUrl = avatarUrl;
    }

    await profile.save();

    success(res, { avatarUrl }, 'Avatar uploaded successfully');
  } catch (err) {
    error(res, err.message, 500);
  }
});

export default router;
```

```javascript
    const { data } = await axios.get(`${API_BASE}/profile`, {

    });
    return data;
  },

  createProfile: async (profileData) => {
    const { data } = await axios.post(`${API_BASE}/profile`, profileData, {
      withCredentials: true
    });
    return data;
  },

  updateProfile: async (profileData) => {
    const { data } = await axios.put(`${API_BASE}/profile`, profileData, {
      withCredentials: true
    });
    return data;
  },

  updatePreferences: async (preferences) => {
    const { data } = await axios.put(`${API_BASE}/profile/preferences`, preferences, {
      withCredentials: true
    });
    return data;
  },

  uploadAvatar: async (file) => {
    const formData = new FormData();
    formData.append('avatar', file);

    const { data } = await axios.post(`${API_BASE}/upload/avatar`, formData, {
      withCredentials: true,
      headers: {
        'Content-Type': 'multipart/form-data'
      }
    });
    return data;
  }
};

export default profileApi;
```

```javascript
import { useState, useEffect } from 'react';
import { profileApi } from '../../api/profile';

export default function ProfileForm({ initialData, onSave }) {
  const [displayName, setDisplayName] = useState(initialData?.displayName || '');
  const [bio, setBio] = useState(initialData?.bio || '');
  const [error, setError] = useState('');
  const [loading, setLoading] = useState(false);

    e.preventDefault();
    setError('');
    setLoading(true);

    try {
      const result = initialData
        ? await profileApi.updateProfile({ displayName, bio })

      if (result.success) {
        onSave(result.data);
      } else {
        setError(result.message || 'Failed to save profile');
      }
    } catch (err) {
      setError(err.response?.data?.message || 'Failed to save profile');
    } finally {
      setLoading(false);
    }
  };

  return (
    <form onSubmit={handleSubmit} className="space-y-4">
      {error && (
        <div className="p-3 bg-red-100 border border-red-400 text-red-700 rounded">
```

```javascript
                <span className="text-gray-500 text-2xl">
                  {profile?.displayName?.charAt(0).toUpperCase() || user.email.charAt(0).toUpperCase()}
                </span>
              </div>
            )}
            <label className="absolute bottom-0 right-0 bg-blue-600 text-white p-1 rounded-full cursor-pointer hover:bg-blue-700">
              <svg className="w-4 h-4" fill="none" stroke="currentColor" viewBox="0 0 24 24">
                <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M15.232 5.232l3.536 3.536m-2.036-5.036a2.5 2.5 0 113.536 3.536L6.5 21.036H3v-3.572L16.732 3.732z" />
              </svg>
              <input
                type="file"
                accept="image/jpeg,image/png,image/webp"
                onChange={handleAvatarChange}
                className="hidden"
              />
            </label>
          </div>
          <div>
            <h2 className="text-xl font-semibold">{profile?.displayName || 'Create your profile'}</h2>
            <p className="text-gray-600">{user.email}</p>
            {profile?.isEmailVerified && (
              <span className="text-xs text-green-600">✓ Email verified</span>
            )}
          </div>
        </div>

        <div className="flex border-b">
          <button
            onClick={() => setActiveTab('profile')}
            className={`px-4 py-2 font-medium ${
              activeTab === 'profile'
                ? 'border-b-2 border-blue-600 text-blue-600'
                : 'text-gray-600 hover:text-gray-800'
            }`}
          >
            Profile
          </button>
          <button
            onClick={() => setActiveTab('preferences')}
            className={`px-4 py-2 font-medium ${
              activeTab === 'preferences'
                ? 'border-b-2 border-blue-600 text-blue-600'
                : 'text-gray-600 hover:text-gray-800'
            }`}
          >
            Synthesis Preferences
          </button>
        </div>

        <div className="mt-4">
          {activeTab === 'profile' && (
            <ProfileForm initialData={profile} onSave={handleProfileSave} />
          )}
          {activeTab === 'preferences' && profile?.synthesisPreferences && (
            <SynthesisPreferences
              initialPreferences={profile.synthesisPreferences}
              onSave={handlePreferencesSave}
            />
          )}
        </div>
      </div>
    </div>
  );
}
```
